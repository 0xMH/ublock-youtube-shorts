! Title: YouTube Shorts Blocker
! Description: Hides all YouTube Shorts surfaces (shelves, cards, sidebar, tabs, chips)
! Homepage: https://github.com/0xMH/ublock-youtube-shorts
! Expires: 7 days
! License: MIT

! YouTube - hide all Shorts (shelves, cards, sidebar, tabs, chips)
youtube.com##ytd-rich-shelf-renderer[is-shorts]
youtube.com##ytd-rich-section-renderer:has(ytd-rich-shelf-renderer[is-shorts])
youtube.com##ytd-reel-shelf-renderer
youtube.com##ytd-item-section-renderer:has(ytd-reel-shelf-renderer)
youtube.com##ytd-video-renderer:has(a[href^="/shorts/"])
youtube.com##ytd-grid-video-renderer:has(a[href^="/shorts/"])
youtube.com##ytd-rich-item-renderer:has(a[href^="/shorts/"])
youtube.com##ytm-shorts-lockup-view-model
youtube.com##ytm-shorts-lockup-view-model-v2
youtube.com##ytd-reel-item-renderer
youtube.com##ytd-guide-entry-renderer:has(a[title="Shorts"])
youtube.com##ytd-mini-guide-entry-renderer[aria-label="Shorts"]
youtube.com##ytd-pivot-bar-item-renderer:has([tab-title="Shorts"])
youtube.com##yt-tab-shape[tab-title="Shorts"]
youtube.com##yt-chip-cloud-chip-renderer:has(yt-formatted-string[title="Shorts"])
youtube.com##yt-chip-cloud-chip-renderer:has-text(/^\s*Shorts\s*$/)
youtube.com##grid-shelf-view-model:has(ytm-shorts-lockup-view-model)
youtube.com##grid-shelf-view-model:has(ytm-shorts-lockup-view-model-v2)

! Catch Shorts disguised as regular videos via the SHORTS thumbnail badge
youtube.com##ytd-video-renderer:has([overlay-style="SHORTS"])
youtube.com##ytd-rich-item-renderer:has([overlay-style="SHORTS"])
youtube.com##ytd-grid-video-renderer:has([overlay-style="SHORTS"])
youtube.com##ytd-compact-video-renderer:has([overlay-style="SHORTS"])

! Mobile-DOM surfaces (ytm-*). Scoped to all subdomains because YouTube
! serves the mobile DOM at www.youtube.com for Android Firefox UAs too.
youtube.com##ytm-pivot-bar-item-renderer:has(.pivot-shorts)
youtube.com##ytm-reel-shelf-renderer
youtube.com##ytm-rich-section-renderer:has(ytm-shorts-lockup-view-model)
youtube.com##ytm-chip-cloud-chip-renderer:has(.yt-core-attributed-string:has-text(/^Shorts$/i))
youtube.com##ytm-video-with-context-renderer:has([data-style="SHORTS"])
